CloudClassifierEngine.java 
package com.udem.classifier.engine;

import com.udem.classifier.model.ClasificacionResult;
import com.udem.classifier.model.CloudModelEnum;
import java.util.ArrayList;
import java.util.List;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * Motor central de lógica de negocio para clasificación de modelos Cloud en Java.
 * Desacoplado de cualquier componente de interfaz gráfica o de consola.
 */
public class CloudClassifierEngine {

    private static final String[] IAAS_PATTERNS = {
        "máquinas?\\s+virtuales?|maquinas?\\s+virtuales?|infraestructura|almacenamiento|redes\\s+configurables|instalar\\s+mi\\s+(propio\\s+)?sistema\\s+operativo|so\\b",
        "servidor(es)?\\s+virtual(es)?|vps|bare\\s+metal|hardware|vpc|subnet|firewall"
    };

    private static final String[] PAAS_PATTERNS = {
        "desplegar\\s+(mi\\s+)?aplicaci[oó]n(\\s+web)?|sin\\s+administrar(\\s+directamente)?\\s+servidores|plataforma|base\\s+de\\s+datos\\s+administrada|esc(ale|ala)\\s+autom[aá]ticamente",
        "app\\s+engine|azure\\s+app\\s+service|elastic\\s+beanstalk|heroku|middleware|runtime"
    };

    private static final String[] SAAS_PATTERNS = {
        "correo\\s+electr[oó]nico|navegador|suscripci[oó]n(\\s+mensual)?|pag(an|a)\\s+una\\s+suscripci[oó]n|software\\s+como\\s+servicio",
        "gmail|google\\s+drive|office\\s+365|salesforce|dropbox|zoom|slack"
    };

    private static final String[] FAAS_PATTERNS = {
        "ejecutar\\s+una\\s+funci[oó]n(\\s+autom[aá]ticamente)?|cada\\s+vez\\s+que\\s+un\\s+usuario\\s+suba|serverless|sin\\s+servidor|funci[oó]n|function",
        "aws\\s+lambda|cloud\\s+functions?|azure\\s+functions?|event-driven|trigger"
    };

    public ClasificacionResult evaluate(String texto) {
        if (texto == null || texto.trim().isEmpty()) {
            return new ClasificacionResult(CloudModelEnum.UNKNOWN, 0, "Texto nulo o vacío", new ArrayList<>());
        }

        String input = texto.toLowerCase();
        List matches = new ArrayList<>();

        int scoreIaaS = countMatches(input, IAAS_PATTERNS, "IaaS", matches);
        int scorePaaS = countMatches(input, PAAS_PATTERNS, "PaaS", matches);
        int scoreSaaS = countMatches(input, SAAS_PATTERNS, "SaaS", matches);
        int scoreFaaS = countMatches(input, FAAS_PATTERNS, "FaaS", matches);

        int max = Math.max(Math.max(scoreIaaS, scorePaaS), Math.max(scoreSaaS, scoreFaaS));

        if (max == 0) {
            return new ClasificacionResult(CloudModelEnum.UNKNOWN, 0, "No se encontraron patrones", matches);
        } else if (max == scoreIaaS) {
            return new ClasificacionResult(CloudModelEnum.IAAS, scoreIaaS, "IaaS - Infrastructure as a Service", matches);
        } else if (max == scorePaaS) {
            return new ClasificacionResult(CloudModelEnum.PAAS, scorePaaS, "PaaS - Platform as a Service", matches);
        } else if (max == scoreSaaS) {
            return new ClasificacionResult(CloudModelEnum.SAAS, scoreSaaS, "SaaS - Software as a Service", matches);
        } else {
            return new ClasificacionResult(CloudModelEnum.FAAS, scoreFaaS, "FaaS - Function as a Service", matches);
        }
    }

    private int countMatches(String text, String[] patterns, String tag, List matches) {
        int count = 0;
        for (String p : patterns) {
            Matcher m = Pattern.compile(p, Pattern.CASE_INSENSITIVE).matcher(text);
            while (m.find()) {
                count++;
                matches.add(tag + ": " + m.group());
            }
        }
        return count;
    }
}

Refactorización y Decoupling de la Arquitectura de Software

Mejoras Arquitectónicas (Single Responsibility Principle)

Se desacopló totalmente la interfaz gráfica (GUI) de la lógica de negocio, siguiendo el patrón de diseño desacoplado (Gamma et al., 1994). Toda la lógica de clasificación se encapsuló en la clase estática CloudClassifier, habilitando la reusabilidad desde interfaces CLI o servicios API REST.

Aplicación al proyecto: main_gui.java crea la raíz Tkinter; gui/main_window.java valida los campos; classifier/cloud_classifier.java calcula el resultado y classification_result.java entrega modelo, puntaje y mensaje. Esta división permite probar la lógica sin abrir la interfaz.

Código Refactorizado (Módulo de Lógica Independiente)

CloudClassifier.java 
import java.util.regex.Pattern;

/**
 * Clase que encapsula exclusivamente la lógica de clasificación.
 * Aplica principios SOLID para evitar acoplamiento con la GUI Swing/CLI.
 */
public class CloudClassifier {

    private static final Pattern IAAS_PATTERN = Pattern.compile(
        "\b(maquina virtual|vm|almacenamiento|red|servidor|infraestructura|escalar|vpc|disco|firewall|balanceador)\b",
        Pattern.CASE_INSENSITIVE
    );
    private static final Pattern PAAS_PATTERN = Pattern.compile(
        "\b(plataforma|desplegar|aplicacion web|sin administrar|runtime|middleware|base de datos administrada|app engine)\b",
        Pattern.CASE_INSENSITIVE
    );
    private static final Pattern SAAS_PATTERN = Pattern.compile(
        "\b(correo|suscripcion|navegador|software|aplicacion|office|gmail|dropbox|salesforce|netflix|spotify)\b",
        Pattern.CASE_INSENSITIVE
    );
    private static final Pattern FAAS_PATTERN = Pattern.compile(
        "\b(funcion|ejecutar automaticamente|evento|trigger|serverless|lambda|cloud functions|response|request)\b",
        Pattern.CASE_INSENSITIVE
    );

    public static String classify(String text) {
        if (text == null || text.trim().isEmpty()) {
            return "No se pudo clasificar (Texto vacío o nulo)";
        }
        String input = text.toLowerCase().trim();

        if (IAAS_PATTERN.matcher(input).find()) return "IaaS";
        if (PAAS_PATTERN.matcher(input).find()) return "PaaS";
        if (SAAS_PATTERN.matcher(input).find()) return "SaaS";
        if (FAAS_PATTERN.matcher(input).find()) return "FaaS";

        return "No se pudo clasificar (Sin coincidencias en el patrón)";
    }
}

Evolución de Reglas Fijas (Regex) hacia Procesamiento de Lenguaje Natural (NLP)

Algoritmo NLP Basado en Lexicón y Scoring

Las expresiones regulares fallan ante la ambigüedad semántica o textos extensos con vocabulario mixto. Por ello, se diseñó un motor de Procesamiento de Lenguaje Natural (NLP) mediante tokenización, remoción de stopwords, stemming algorítmico y asignación de pesos ponderados por categoría (Jurafsky & Martin, 2023).

  1. Normalización: convertir el texto a minúsculas y retirar signos para que “Máquinas” y “maquinas” se traten igual.
  2. Tokenización: separar la descripción en palabras analizables.
  3. Filtrado: eliminar stopwords como “de”, “la” y “para”, que no distinguen un modelo Cloud.
  4. Stemming y puntaje: reducir variantes como “desplegar/desplegando” y sumar sus pesos por categoría.
  5. Decisión: retornar el modelo con mayor score, o Desconocido cuando no hay coincidencias.
CloudClassifierNLP.java 
import java.util.*;
import java.util.stream.Collectors;

/**
 * Clasificador mejorado utilizando técnicas fundamentales de NLP.
 */
public class CloudClassifierNLP {

    private static final Set STOPWORDS = new HashSet<>(Arrays.asList(
        "de", "la", "que", "el", "en", "y", "a", "los", "del", "las", "un", "por", "con", "no", "su", "para", "es", "lo", "como"
    ));

    private static final Map IAAS_WEIGHTS = Map.of(
        "maquin", 3, "virtual", 3, "almacen", 2, "red", 2, "servidor", 2, "infraestructur", 3, "vpc", 3
    );
    private static final Map PAAS_WEIGHTS = Map.of(
        "plataform", 3, "despleg", 3, "runtime", 3, "middleware", 3, "bas", 2, "dat", 1
    );
    private static final Map SAAS_WEIGHTS = Map.of(
        "corre", 3, "suscripci", 3, "navegad", 2, "softwar", 3, "gmail", 3, "office", 3
    );
    private static final Map FAAS_WEIGHTS = Map.of(
        "funci", 3, "event", 3, "trigger", 3, "serverless", 3, "lambda", 3, "automatic", 2
    );

    public static String classify(String text) {
        if (text == null || text.trim().isEmpty()) return "Entrada inválida";

        String clean = text.toLowerCase().replaceAll("[^a-záéíóúñ\s]", " ");
        String[] tokens = clean.split("\s+");

        List filtered = Arrays.stream(tokens)
            .filter(w -> !STOPWORDS.contains(w) && w.length() > 2)
            .collect(Collectors.toList());

        int scoreIaaS = 0, scorePaaS = 0, scoreSaaS = 0, scoreFaaS = 0;

        for (String word : filtered) {
            String stem = word.replaceAll("(ando|iendo|mente|cion|s|es)$", "");
            scoreIaaS += IAAS_WEIGHTS.getOrDefault(stem, 0);
            scorePaaS += PAAS_WEIGHTS.getOrDefault(stem, 0);
            scoreSaaS += SAAS_WEIGHTS.getOrDefault(stem, 0);
            scoreFaaS += FAAS_WEIGHTS.getOrDefault(stem, 0);
        }

        int max = Math.max(Math.max(scoreIaaS, scorePaaS), Math.max(scoreSaaS, scoreFaaS));
        if (max == 0) return "No clasificado (Sin coincidencia NLP)";

        if (max == scoreIaaS) return "IaaS";
        if (max == scorePaaS) return "PaaS";
        if (max == scoreSaaS) return "SaaS";
        return "FaaS";
    }
}

NLP, Regex y procesamiento

El pipeline usa UTF-8, pasa el texto a minúsculas, elimina tildes con unicodedata.normalize('NFKD'), tokeniza con re.findall(r'\\b\\w+\\b', texto), filtra stopwords y aplica stemming básico. El clasificador suma pesos por coincidencia y selecciona el modelo con puntaje máximo, por lo que GUI y CLI comparten exactamente la misma lógica.

classifier/nlp_processor.py 
import re
import unicodedata
from .stopwords import STOPWORDS

class NLPProcessor:
    """Procesador de Lenguaje Natural para normalización y limpieza de texto."""

    def _normalize_text(self, text: str) -> str:
        """Elimina acentos y diacríticos."""
        nfkd = unicodedata.normalize('NFKD', text)
        return "".join([c for c in nfkd if not unicodedata.combining(c)])

    def _tokenize(self, text: str) -> list[str]:
        """Divide el texto en tokens alfanuméricos."""
        return re.findall(r'\w+', text)

    def _clean_and_filter(self, tokens: list[str]) -> list[str]:
        """Filtra palabras vacías (stopwords) y de corta longitud."""
        return [w for w in tokens if w not in STOPWORDS and len(w) > 2]

    def _apply_stemming(self, tokens: list[str]) -> list[str]:
        """Aplica sufijación básica en español."""
        stemmed = []
        for word in tokens:
            s = re.sub(r'(aciones|acion|es|s|es|mente|ando|iendo|idades|idad)$', '', word)
            stemmed.append(s)
        return stemmed

    def process_text(self, text: str) -> list[str]:
        """Pipeline completo de NLP."""
        if not text:
            return []
        lower = text.lower()
        normalized = self._normalize_text(lower)
        tokens = self._tokenize(normalized)
        cleaned = self._clean_and_filter(tokens)
        return self._apply_stemming(cleaned)

from .nlp_processor import NLPProcessor
from .classification_result import ClassificationResult
from .cloud_model import CloudModel

class CloudClassifier:
    """Motor de clasificación de modelos Cloud mediante Scoring Ponderado."""

    def __init__(self):
        self.nlp = NLPProcessor()
        self.terms = {
            CloudModel.IAAS: {
                'virt': 5, 'serv': 4, 'infra': 5, 'ec2': 5, 'storage': 4,
                'almacen': 4, 's3': 5, 'vpc': 4, 'load': 3, 'balancer': 3,
                'firewall': 3, 'disk': 3, 'escal': 3, 'comput': 4
            },
            CloudModel.PAAS: {
                'plat': 5, 'platform': 5, 'heroku': 5, 'app': 4, 'aplic': 4,
                'deploy': 4, 'despleg': 4, 'runtime': 3, 'middleware': 3,
                'databas': 3, 'basedat': 3, 'firebase': 4, 'beanstalk': 4
            },
            CloudModel.SAAS: {
                'corre': 5, 'email': 5, 'suscripci': 5, 'softwar': 5, 'gmail': 5,
                'office': 5, 'dropbox': 5, 'salesforc': 5, 'navegad': 3, 'browser': 3
            },
            CloudModel.FAAS: {
                'funci': 5, 'func': 5, 'lambda': 5, 'event': 4, 'trigger': 4,
                'serverless': 5, 'autom': 4, 'cloudfuncti': 5, 'ejecut': 3
            }
        }

    def classify(self, text: str) -> ClassificationResult:
        tokens = self.nlp.process_text(text)
        if not tokens:
            return ClassificationResult(CloudModel.UNKNOWN, 0.0, "Texto vacío")

        scores = {model: 0 for model in CloudModel if model != CloudModel.UNKNOWN}

        for token in tokens:
            for model, term_map in self.terms.items():
                for term, weight in term_map.items():
                    if term in token or token in term:
                        scores[model] += weight

        best_model = max(scores, key=scores.get)
        max_score = scores[best_model]

        if max_score == 0:
            return ClassificationResult(CloudModel.UNKNOWN, 0.0, "Sin coincidencia")

        return ClassificationResult(best_model, float(max_score), "Clasificación exitosa")

Análisis Detallado sobre la Posición Estratégica de FaaS

FaaS (Function as a Service) representa el pináculo del modelo Serverless (Castro et al., 2019). A diferencia de PaaS, donde la aplicación se ejecuta de forma continua en una instancia o contenedor reservado, FaaS es puramente conducido por eventos (Event-Driven). La infraestructura no existe para el desarrollador hasta que un evento déclenche la ejecución, garantizando cero costo en estado inactivo (scale-to-zero).

¿Cloud-Native o Cloud-Enabled? Análisis Arquitectónico Profundo de la Plataforma Global de Streaming Spotify

Spotify (2026)

Resumen Ejecutivo

El presente ensayo realiza un análisis técnico exhaustivo sobre la arquitectura distribuida y la infraestructura de cómputo de la plataforma global de audio en streaming Spotify. A través de la descomposición formal de sus capas operativas y la revisión de su migración estratégica desde centros de datos colocalizados hacia Google Cloud Platform (GCP), se argumenta fundadamente por qué Spotify constituye un caso emblemático de arquitectura puramente Cloud-Native. El documento evalúa detalladamente los pilares de su ecosistema: patrones de microservicios, orquestación con Kubernetes, interfaces API gRPC/REST, cómputo serverless (FaaS), bases de datos NoSQL administradas, observabilidad continua, Infraestructura como Código (IaC) y estrategias de alta disponibilidad y tolerancia a fallos (Izrailevsky, 2016; Kratzke & Quint, 2017).

1. Introducción y Marco Teórico

En la ingeniería de software moderna, resulta imprescindible distinguir con rigor formal entre los paradigmas Cloud-Enabled y Cloud-Native al evaluar la madurez tecnológica de un sistema (CNCF, 2023). Un software considerado Cloud-Enabled (o "re-hosted") corresponde a una aplicación construida originalmente bajo paradigmas tradicionales u on-premises que ha sido migrada a entornos virtuales en la nube mediante un enfoque "Lift-and-Shift" (Jain, 2014). Aunque estas aplicaciones aprovechan la infraestructura como servicio (IaaS), conservan su arquitectura monolítica, acoplamientos rígidos entre componentes, estados persistentes en memoria local y una fuerte dependencia de parches manuales de infraestructura (Bass et al., 2021).

Por el contrario, un sistema Cloud-Native se diseña desde sus cimientos para maximizar la velocidad, la escalabilidad elástica y la resiliencia en entornos de nube pública dinámicos y distribuidos (Microsoft Learn, 2026; Fowler, 2015). De acuerdo con la Cloud Native Computing Foundation (CNCF, 2023), los sistemas Cloud-Native se caracterizan por el empaquetado de procesos en contenedores aislados, la orquestación dinámica de cargas de trabajo, la descomposición en microservicios desacoplados y la automatización estricta del ciclo de vida del software a través de prácticas DevOps, pruebas continuas e Infraestructura como Código (IaC) (Wiggins, 2017).

2. Aplicación Seleccionada: Spotify

Spotify es la plataforma de transmisión de música, podcasts y audiolibros líder a nivel mundial, prestando servicio a más de 600 millones de usuarios activos mensuales y procesando decenas de miles de millones de eventos diarios (Spotify Engineering, 2024). En sus inicios (2006-2015), Spotify operaba su propia infraestructura física distribuida en cuatro data centers en Dallas, Londres, Estocolmo y San José. Sin embargo, el crecimiento acelerado de la base de usuarios generó fricciones operativas masivas en la provisión de hardware. Entre 2015 y 2017, la empresa ejecutó una histórica reingeniería arquitectónica para trasladar la totalidad de su carga transaccional y analítica a Google Cloud Platform (GCP), rediseñando sus componentes para operar nativamente en la nube (GCP Stories, 2021).

3. Análisis Tecnológico Detallado

El éxito operativo de Spotify se fundamenta en una rigurosa implementación de patrones de arquitectura distribuida evaluados bajo los siguientes pilares:

  • Arquitectura Monolítica vs. Microservicios: Spotify trascendió la estructura monolítica tradicional dividiendo su sistema en una malla de más de 1,500 microservicios independientes y autónomos (Nygren, 2022). Cada microservicio es propiedad de un equipo multidisciplinario pequeño ("Squad") que gestiona de manera aislada su ciclo de vida, lógica de negocio y esquema de despliegue. Esta descomposición evita los cuellos de botella en la compilación y garantiza una agilidad organizacional superior.
  • Contenedores y Orquestación: Históricamente, Spotify desarrolló un orquestador propietario denominado Helios. Sin embargo, para estandarizarse con la industria, migró por completo su flota de servicios a contenedores Docker gestionados mediante Google Kubernetes Engine (GKE) (Kubernetes Cases, 2022). GKE gestiona automáticamente el aprovisionamiento de pods, la programación de cargas de trabajo y la auto-recuperación de contenedores caídos sin intervención humana.
  • Gestión de APIs y Comunicación: La interacción entre microservicios no se realiza mediante llamadas directas acopladas, sino a través de contratos de API estrictos. Para la comunicación interna de baja latencia inter-servicio, Spotify emplea gRPC con Protocol Buffers, reduciendo drásticamente la sobrecarga de serialización (Nygren, 2022). Para la comunicación asíncrona guiada por eventos, utiliza clústeres masivos de Apache Kafka como bus de mensajes pub/sub.
  • Serverless / FaaS: Para tareas efímeras, procesamiento de archivos multimedia y flujos conducidos por eventos que no justifican una instancia de contenedor ejecutándose continuamente, Spotify integra servicios Serverless como Google Cloud Functions y Cloud Run (Castro et al., 2019). Esto permite escalar el cómputo instantáneamente a cero cuando no hay demanda, optimizando los costos operativos.
  • Bases de Datos Administradas: Abandonando las bases de datos relacionales únicas que representaban un punto único de falla, Spotify implementó un enfoque de *Polyglot Persistence* con bases de datos administradas en GCP (Google Cloud, 2026b). Utiliza Cloud Bigtable para el almacenamiento de baja latencia de historiales de reproducción y sistemas de recomendación, DataStax Cassandra gestionado para catálogos de metadatos NoSQL distribuida, y Cloud Spanner para transacciones financieras que requieren consistencia ACID global.
  • Escalabilidad Horizontal y Elasticidad: La infraestructura aprovecha las capacidades de Horizontal Pod Autoscaler (HPA) de Kubernetes y autoscaling de nodos en GKE. Ante picos globales de tráfico (por ejemplo, el lanzamiento de un álbum masivo o el evento anual *Spotify Wrapped*), el sistema incrementa elásticamente réplicas de pods y nodos de cómputo en segundos, contrayéndose automáticamente cuando la carga disminuye.
  • Alta Disponibilidad: La plataforma se despliega de forma multirregional a través de múltiples zonas de disponibilidad (*Availability Zones*) de Google Cloud Platform. La distribución geográfica del tráfico mediante BGP Anycast garantiza que, ante la caída catastrófica de un data center completo de GCP, el tráfico se redirija transparentemente a otra región sin interrupción del servicio para el oyente.
  • Observabilidad Avanzada: La salud de la arquitectura distribuida se monitorea en tiempo real recopilando métricas, trazabilidad distribuida y logs centralizados. Spotify utiliza herramientas avanzadas de trazado como OpenTelemetry y colectores de telemetría que permiten rastrear una única petición de usuario a través de decenas de microservicios mediante IDs de correlación únicos (Spotify Engineering, 2024).
  • DevOps y CI/CD: Spotify es pionero mundial en cultura DevOps. La empresa creó y donó a la CNCF la plataforma Backstage.io, un portal de desarrolladores que estandariza la creación de microservicios, documentación e integración continua (Backstage.io, 2024). Cada cambio en el código se valida mediante pipelines automáticos de CI/CD ejecutados en minutos, permitiendo miles de despliegues diarios a producción con cero tiempo de inactividad (*Canary Deployments*).
  • Infraestructura como Código (IaC): Toda la infraestructura de servidores, reglas de red, políticas de IAM y clústeres de GKE está declarada en archivos de configuración versiónados administrados con Terraform (Bass et al., 2021). Ningún entorno se configura manualmente vía consola web, garantizando la repetibilidad exacta de los entornos de desarrollo, staging y producción.
  • Tolerancia a Fallos: Al operar miles de servicios en la nube, Spotify asume que las fallas de hardware y red son inevitables. Por ello, la arquitectura implementa patrones de resiliencia como Circuit Breaker, Bulkheads y degradación elegante (*Graceful Degradation*). Si el microservicio de recomendaciones personalizadas falla, la aplicación no colapsa; simplemente muestra listas de reproducción populares locales almacenadas en caché (Spotify Engineering, 2024).

4. Argumentación Categoría Cloud-Native vs. Cloud-Enabled

Al analizar de manera integral la plataforma de Spotify, queda demostrado que la empresa **no se encuadra en la categoría Cloud-Enabled**. Un enfoque *Cloud-Enabled* se habría limitado a virtualizar la arquitectura antigua en instancias IaaS (como Compute Engine) manteniendo sus servidores monolíticos y procesos de despliegue manuales.

Por el contrario, Spotify califica indiscutiblemente como una plataforma **Cloud-Native**. La compañía reestructuró la totalidad de sus aplicaciones para alinearse con los Doce Factores (*Twelve-Factor App*) (Wiggins, 2017), opera completamente sobre contenedores declarativos en Kubernetes, automatiza la infraestructura con Terraform, utiliza servicios administrados Serverless/NoSQL y soporta despliegues continuos desacoplados por microservicios.

5. Conclusiones Académicas

El estudio de caso de Spotify demuestra que la transición hacia la nube no representa simplemente un cambio en el lugar de alojamiento físico del código, sino una transformación profunda en los patrones de diseño arquitectónico y en la cultura organizacional. La adopción del paradigma Cloud-Native ha sido el factor tecnológico clave que ha permitido a Spotify escalar su negocio globalmente, soportando millones de usuarios concurrentes con una disponibilidad del 99.99% y liberando nuevas funcionalidades al mercado a una velocidad sin precedentes en la industria del entretenimiento.

6. Conclusiones Personales

La solución debe equilibrar la agilidad, alta disponibilidad y elasticidad de la nube pública con las estrictas regulaciones financieras de soberanía de datos y seguridad (PCI-DSS v4.0, CNBV, ISO/IEC 27001) que obligan a mantener el Core Bancario transaccional y los módulos criptográficos HSM en un entorno On-Premises (Bass et al., 2021).

Herramienta de IA Generativa Utilizada: Cursor IDE (potenciado con el modelo DeepSeek-R1 / Claude 3.5 Sonnet), utilizado como agente de arquitectura de software e ingeniería de código para prototipado rápido, auditoría de vulnerabilidades y refactorización continua.

2. Decisiones de Distribución Híbrida (On-Premises vs. Cloud)

Capa / Componente Ubicación Justificación Arquitectónica y Cumplimiento Normativo
Core Bancario & Ledger de Cuentas On-Premises (Mainframe IBM z16 / DB2) Cumplimiento estricto de soberanía de datos financieros, latencia ultrabaja en liquidación y aislamiento de red.
Módulos Criptográficos HSM On-Premises (Hardware Criptográfico Thales) Resguardo físico de claves maestras (PIN/PVK) para firma de transacciones interbancarias.
WAF & Protección DDoS Perimetral Nube Pública (AWS WAF / CloudFront) Filtrado masivo de tráfico malicioso en el borde (Edge) y mitigación de ataques OWASP Top 10 antes de tocar la red local.
API Gateway & Autenticación Nube Pública (AWS API Gateway + Cognito) Escalabilidad elástica en la emisión de tokens JWT, autenticación multifactor (MFA) y enrutamiento inteligente.
Alertas & Notificaciones FaaS Nube Pública (AWS Lambda + Amazon SNS) Procesamiento serverless guiado por eventos para envío instantáneo de notificaciones Push, SMS y correos transaccionales.
Monitoreo y Observabilidad Híbrido (Datadog / Dynatrace) Trazabilidad end-to-end unificada mediante IDs de correlación entre eventos Cloud y transacciones On-Premises.

3. Proceso de Prompt Engineering e Iteración Continuada

Para no aceptar de forma pasiva la primera propuesta de la IA, se ejecutó un proceso iterativo de 5 Prompts de Mejora, evaluando y corrigiendo técnicamente la solución en cada paso:

Prompt 0: Inicial (Propuesta de Partida) 
Actúa como arquitecto de soluciones Cloud. Diseña una arquitectura para una aplicación de banca móvil utilizando una estrategia de nube híbrida. La solución deberá incluir aplicación móvil, API Gateway, servicios backend, autenticación, bases de datos, almacenamiento, funciones serverless/FaaS, monitoreo y mecanismos de seguridad. Identifica qué componentes deberían permanecer on-premises y cuáles deberían ejecutarse en la nube, justificando cada decisión. Propón una arquitectura escalable, tolerante a fallos y altamente disponible. Genera un diagrama de arquitectura y desarrolla un prototipo básico del backend en Python.
Prompt 1 de Mejora: Seguridad y Módulo de Auditoría 
Revisa la arquitectura propuesta desde la perspectiva de ciberseguridad financiera (OWASP Mobile & API). Detecta vulnerabilidades en la comunicación entre la Nube y el On-Premises. Agrega un módulo de auditoría de seguridad automatizado en el backend de Python que verifique en tiempo real el uso de cifrado TLS 1.3, validación de tokens JWT sin datos sensibles expuestos y ausencia de llaves hardcodeadas.

Avance / Corrección 1: La IA inicialmente propuso conexiones HTTP simples para pruebas. Se forzó la implementación de conexiones tunelizadas VPN IPSec / AWS Direct Connect con cifrado obligatorio TLS 1.3 y verificación estricta de políticas CORS en el API Gateway.

Prompt 2 de Mejora: Reglas de Negocio Financieras (Saldo Cero) 
La propuesta actual permite que un usuario intente realizar transferencias sin validar previamente su saldo. Modifica el código del backend y de la GUI en Python para que, si el cliente tiene $0.00 de saldo disponible, la aplicación bloquee e inhabilite automáticamente los botones y endpoints de transferencia, notificando al usuario y evitando llamadas innecesarias al API Gateway.

Avance / Corrección 2: Se desacopló la lógica de validación previa en el cliente móvil y en la capa Edge, reduciendo la carga de peticiones inútiles hacia el Core Bancario On-Premises cuando la cuenta no tiene fondos.

Prompt 3 de Mejora: Recuperación ante Desastres (DR) y Doble Confirmación 
Incorpora mecanismos explícitos de Recuperación ante Desastres (Disaster Recovery). En el flujo de transferencia, implementa un diálogo de seguridad de doble confirmación activa donde el usuario deba validar los datos antes del procesamiento interbancario, e imita una conmutación por error (Failover) si el nodo primario de la API no responde.

Avance / Corrección 3: Se añadió una capa de resiliencia tipo Circuit Breaker en la API para redirigir peticiones a un sitio secundario (DR) en menos de 3 segundos ante una eventual caída de la región principal.

Prompt 4 de Mejora: Visibilidad de API y Logs de Notificación 
En el prototipo actual no es evidente cuándo se están realizando llamadas a las APIs Cloud. Agrega a la interfaz gráfica y consola un panel de notificación en tiempo real que registre los eventos, handshakes criptográficos, llamadas al API Gateway y notificaciones FaaS emitidas con sus respectivos timestamps.

Avance / Corrección 4: Se integró una consola interactiva de logs de auditoría dentro de la GUI Tkinter para visualizar de manera clara el flujo de comunicación entre el frontend móvil, el API Gateway y el backend On-Premises.

Prompt 5 de Mejora: Suite Automatizada de Pruebas Unitarias 
Para garantizar la calidad y validación técnica de la solución, genera una suite de 10 pruebas unitarias automáticas en un archivo `test_cases.py` que evalúe login exitoso/fallido, consulta de saldo, bloqueo de saldo cero, confirmación DR, conectividad TLS 1.3, auditoría de seguridad y conmutación de failover.

Avance / Corrección 5: Se construyó una suite de testing ejecutable que automatiza la verificación de requisitos funcionales y no funcionales del sistema bancario.

    def process_transfer(self):
        if self.balance <= 0:
            messagebox.showerror("Error", "No cuenta con saldo disponible.")
            return

        try:
            monto = float(self.ent_amount.get().strip())
            dest = self.ent_dest.get().strip()
            concepto = self.ent_concept.get().strip()

            if monto <= 0 or monto > self.balance:
                messagebox.showerror("Monto Inválido", f"El monto debe ser mayor a $0 y no superar su saldo de ${self.balance:,.2f}.")
                return

            confirm = messagebox.askyesno("Confirmación Criptográfica (DR)", f"¿Confirma transferir ${monto:,.2f} MXN a la cuenta {dest}?\nFirma HSM On-Premises requerida.")
            if confirm:
                self.balance -= monto
                self.lbl_balance.config(text=f"${self.balance:,.2f} MXN")
                res = self.api_client.post_transaction(monto)
                self.log(f"[{datetime.datetime.now().strftime('%H:%M:%S')}] {res['message']} | Concepto: {concepto}")
                messagebox.showinfo("Transferencia Exitosa", f"Transferencia de ${monto:,.2f} procesada correctamente.")
                self.check_rules_transfer()

        except ValueError:
            messagebox.showerror("Error de Formato", "Ingrese un monto numérico válido.")

Evolución mediante Prompt Engineering: La interacción con Cursor IDE demostró que el diseño de arquitecturas complejas mediante IA no es un evento único, sino un proceso iterativo de refinamiento técnico. La primera propuesta generada por la IA ofrecía una vista superficial y genérica de la nube híbrida, obviando aspectos clave de ciberseguridad financiera y rendimiento. A través del uso de al menos cinco prompts de mejora dirigidos, se logró transformar un borrador simplista en un diseño bancario robusto con tunelización VPN Direct Connect, autenticación multifactor, monitoreo centralizado y resiliencia ante desastres.

Evaluación Crítica de la Arquitectura y Omisiones Detectadas: La propuesta inicial de la IA presentaba fallas y omisiones cuestionables. Por ejemplo, sugería enviar transacciones directamente desde la aplicación móvil hacia el backend On-Premises mediante peticiones HTTP convencionales sin filtrar en el borde. Tampoco contempló de origen la regla de negocio crítica de invalidar las transferencias cuando el usuario tiene saldo cero, lo que habría provocado una saturación innecesaria de peticiones hacia el mainframe corporativo. Estas decisiones habrían expuesto a la institución a ataques de denegación de servicio (DDoS) y vulnerabilidades serias de seguridad.

Análisis de Seguridad y Contexto Financiero: Las herramientas de IA generativa carecen de un entendimiento contextual profundo sobre las regulaciones financieras locales e internacionales (como PCI-DSS v4.0 o las normativas de la CNBV). Aunque la IA conoce los términos teóricos como "cifrado" o "autenticación", no aplica por defecto mecanismos strictly como la firma de transacciones mediante hardware HSM On-Premises o la gestión de llaves maestras en entornos aislados. Fue necesaria la intervención activa del estudiante para guiar a la IA e imponer el uso de cifrado TLS 1.3 sin excepciones.

Comprensión del Código y Validación: El prototipo backend y la GUI en Python generados fueron analizados y refactorizados de forma transparente. Comprendo completamente la estructura limpia utilizada, la lógica de validación previa de saldo, la emisión de eventos de log y el uso de clientes mockeados para simular handshakes criptográficos. La validación se llevó a cabo no solo mediante inspección visual, sino ejecutando una suite automatizada de 10 pruebas unitarias que verificó cada escenario límite (edge cases).

El Rol del Ingeniero en Tecnologías Computacionales (ITC): Un Ingeniero en Tecnologías Computacionales no puede ser reemplazado por un agente de IA generativa. Para evaluar, validar y corregir una arquitectura propuesta por IA, el ingeniero requiere bases sólidas y multidisciplinarias en ingeniería de software, patrones de diseño, redes de computadoras, ciberseguridad, gobernanza de datos y bases de datos transaccionales. Se pueden delegar a la IA tareas repetitivas como la generación de código boilerplate, la sintaxis de scripts o borradores iniciales de diagramas; sin embargo, la responsabilidad final de las decisiones arquitectónicas, la auditoría de seguridad, el cumplimiento regulatorio y la garantía de resiliencia financiera debe permanecer única y exclusivamente bajo la responsabilidad humana.

Conclusión: Una persona sin bases sólidas de arquitectura, programación, redes y ciberseguridad jamás podría evaluar correctamente una solución bancaria generada por IA. Aceptar ciegamente las respuestas de un LLM en un entorno crítico derivaría en arquitecturas ineficientes, costosas y extremadamente vulnerables a ciberataques o pérdidas financieras catastróficas.

10. Conclusiones de la Tarea 4

La realización de esta tarea permitió aplicar la IA como una herramienta avanzada de co-diseño e ingeniería de software, demostrando que el valor profesional del ingeniero reside en su capacidad de juicio crítico, verificación y refinamiento. La solución final lograda equilibra las exigencias normativas del sector bancario con los beneficios de escalabilidad de la nube pública, proporcionando un modelo de referencia seguro, resiliente y de alto rendimiento.
